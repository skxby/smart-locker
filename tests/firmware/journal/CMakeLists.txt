cmake_minimum_required(VERSION 3.16)
project(locker_journal_tests C)
enable_testing()
add_library(journal_littlefs STATIC
    ../../../firmware/third_party/littlefs/lfs.c
    ../../../firmware/third_party/littlefs/lfs_util.c)
target_include_directories(journal_littlefs PUBLIC ../../../firmware/third_party/littlefs)
target_compile_definitions(journal_littlefs PUBLIC LFS_NO_MALLOC LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
add_executable(test_journal test_journal.c nor_fixture.c ../../../firmware/journal/journal.c)
target_include_directories(test_journal PRIVATE ../../../firmware/journal ../../../contracts/include)
target_compile_options(test_journal PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
set_target_properties(test_journal PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
target_link_libraries(test_journal PRIVATE journal_littlefs)
add_test(NAME firmware.journal_transactions_and_powercuts COMMAND test_journal)
if(UNIX)
    find_package(Threads REQUIRED)
    add_executable(test_journal_concurrency test_concurrency.c nor_fixture.c
        ../../../firmware/journal/journal.c)
    target_include_directories(test_journal_concurrency PRIVATE ../../../firmware/journal ../../../contracts/include)
    target_compile_options(test_journal_concurrency PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    set_target_properties(test_journal_concurrency PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    target_link_libraries(test_journal_concurrency PRIVATE journal_littlefs Threads::Threads)
    add_test(NAME firmware.journal_concurrent_owner COMMAND test_journal_concurrency)
    set_tests_properties(firmware.journal_concurrent_owner PROPERTIES TIMEOUT 30)
endif()

    include(flash-test.cmake)
endif()
